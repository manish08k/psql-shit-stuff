# 03 — ALTER (all objects)

## ALTER TABLE
```sql
ALTER TABLE [IF EXISTS] [ONLY] t RENAME TO t2;
ALTER TABLE t RENAME COLUMN a TO b;
ALTER TABLE t RENAME CONSTRAINT c1 TO c2;
ALTER TABLE t SET SCHEMA s;
ALTER TABLE t OWNER TO r;
ALTER TABLE t SET TABLESPACE ts;
ALTER TABLE t SET LOGGED | UNLOGGED;

-- columns
ALTER TABLE t ADD COLUMN [IF NOT EXISTS] c type [DEFAULT x] [NOT NULL] [constraint];
ALTER TABLE t DROP COLUMN [IF EXISTS] c [CASCADE];
ALTER TABLE t ALTER COLUMN c TYPE newtype [USING c::newtype];
ALTER TABLE t ALTER COLUMN c SET DEFAULT x;
ALTER TABLE t ALTER COLUMN c DROP DEFAULT;
ALTER TABLE t ALTER COLUMN c SET NOT NULL;
ALTER TABLE t ALTER COLUMN c DROP NOT NULL;
ALTER TABLE t ALTER COLUMN c SET EXPRESSION AS (expr);          -- generated columns
ALTER TABLE t ALTER COLUMN c DROP EXPRESSION;
ALTER TABLE t ALTER COLUMN c ADD GENERATED ALWAYS AS IDENTITY;
ALTER TABLE t ALTER COLUMN c SET GENERATED BY DEFAULT;
ALTER TABLE t ALTER COLUMN c RESTART WITH 1000;
ALTER TABLE t ALTER COLUMN c DROP IDENTITY [IF EXISTS];
ALTER TABLE t ALTER COLUMN c SET STATISTICS 500;
ALTER TABLE t ALTER COLUMN c SET STORAGE PLAIN|EXTERNAL|EXTENDED|MAIN;
ALTER TABLE t ALTER COLUMN c SET COMPRESSION pglz|lz4;

-- constraints
ALTER TABLE t ADD CONSTRAINT pk PRIMARY KEY (id);
ALTER TABLE t ADD CONSTRAINT uq UNIQUE (a, b);
ALTER TABLE t ADD CONSTRAINT ck CHECK (x > 0) NOT VALID;
ALTER TABLE t VALIDATE CONSTRAINT ck;
ALTER TABLE t ADD CONSTRAINT fk FOREIGN KEY (x) REFERENCES o(id) ON DELETE SET NULL;
ALTER TABLE t ADD PRIMARY KEY USING INDEX idx;
ALTER TABLE t DROP CONSTRAINT [IF EXISTS] c [CASCADE];
ALTER TABLE t ALTER CONSTRAINT c DEFERRABLE INITIALLY DEFERRED;

-- triggers / rules / RLS
ALTER TABLE t ENABLE|DISABLE TRIGGER trg|ALL|USER;
ALTER TABLE t ENABLE ALWAYS|REPLICA TRIGGER trg;
ALTER TABLE t ENABLE|DISABLE RULE r;
ALTER TABLE t ENABLE|DISABLE ROW LEVEL SECURITY;
ALTER TABLE t FORCE|NO FORCE ROW LEVEL SECURITY;

-- misc
ALTER TABLE t SET (fillfactor=70, autovacuum_enabled=false);
ALTER TABLE t RESET (fillfactor);
ALTER TABLE t CLUSTER ON idx;
ALTER TABLE t SET WITHOUT CLUSTER;
ALTER TABLE t REPLICA IDENTITY DEFAULT|FULL|NOTHING|USING INDEX idx;
ALTER TABLE t INHERIT parent;   ALTER TABLE t NO INHERIT parent;
ALTER TABLE t OF type;          ALTER TABLE t NOT OF;

-- partitions
ALTER TABLE m ATTACH PARTITION p FOR VALUES FROM (1) TO (10);
ALTER TABLE m DETACH PARTITION p [CONCURRENTLY | FINALIZE];
ALTER TABLE m MERGE PARTITIONS (p1, p2) INTO p12;
ALTER TABLE m SPLIT PARTITION p INTO (PARTITION a FOR VALUES ..., PARTITION b DEFAULT);
```

## Database / schema / tablespace
```sql
ALTER DATABASE db RENAME TO db2;
ALTER DATABASE db OWNER TO r;
ALTER DATABASE db SET TABLESPACE ts;
ALTER DATABASE db SET param = value;            ALTER DATABASE db RESET param;
ALTER DATABASE db WITH ALLOW_CONNECTIONS false CONNECTION LIMIT 10 IS_TEMPLATE true;
ALTER DATABASE db REFRESH COLLATION VERSION;
ALTER SCHEMA s RENAME TO s2;                    ALTER SCHEMA s OWNER TO r;
ALTER TABLESPACE ts RENAME TO ts2;              ALTER TABLESPACE ts OWNER TO r;
ALTER TABLESPACE ts SET (random_page_cost = 1.1);
```

## Roles / users / groups
```sql
ALTER ROLE r WITH PASSWORD 'pw' LOGIN CREATEDB SUPERUSER VALID UNTIL 'infinity' CONNECTION LIMIT 5;
ALTER ROLE r RENAME TO r2;
ALTER ROLE r [IN DATABASE db] SET param = value;
ALTER ROLE r RESET ALL;
ALTER ROLE ALL SET search_path = public;
ALTER USER u PASSWORD 'pw';
ALTER GROUP g ADD USER u1, u2;                  ALTER GROUP g DROP USER u1;
ALTER GROUP g RENAME TO g2;
ALTER DEFAULT PRIVILEGES [FOR ROLE r] [IN SCHEMA s] GRANT SELECT ON TABLES TO u;
ALTER DEFAULT PRIVILEGES REVOKE EXECUTE ON FUNCTIONS FROM PUBLIC;
ALTER POLICY p ON t RENAME TO p2;               ALTER POLICY p ON t TO role USING (expr) WITH CHECK (expr);
```

## Indexes / views / sequences
```sql
ALTER INDEX idx RENAME TO idx2;
ALTER INDEX idx SET TABLESPACE ts;
ALTER INDEX idx SET (fillfactor=80);
ALTER INDEX idx ATTACH PARTITION idx_part;
ALTER INDEX idx ALTER COLUMN 1 SET STATISTICS 500;
ALTER INDEX idx DEPENDS ON EXTENSION ext;
ALTER VIEW v ALTER COLUMN c SET DEFAULT x;      ALTER VIEW v RENAME TO v2;
ALTER VIEW v OWNER TO r;                        ALTER VIEW v SET SCHEMA s;
ALTER VIEW v SET (security_invoker = true);
ALTER MATERIALIZED VIEW mv RENAME TO mv2;       ALTER MATERIALIZED VIEW mv SET TABLESPACE ts;
ALTER MATERIALIZED VIEW mv RENAME COLUMN a TO b;
ALTER SEQUENCE seq RESTART WITH 1 INCREMENT BY 2 MINVALUE 1 MAXVALUE 100 CACHE 10 [NO] CYCLE OWNED BY t.col;
ALTER SEQUENCE seq SET LOGGED|UNLOGGED;         ALTER SEQUENCE seq RENAME TO s2;
```

## Types / functions / operators / misc objects
```sql
ALTER TYPE t RENAME TO t2;                      ALTER TYPE t OWNER TO r;       ALTER TYPE t SET SCHEMA s;
ALTER TYPE mood ADD VALUE 'meh' BEFORE|AFTER 'ok';
ALTER TYPE mood RENAME VALUE 'meh' TO 'so-so';
ALTER TYPE comp ADD ATTRIBUTE a int;            ALTER TYPE comp DROP ATTRIBUTE a;
ALTER TYPE comp ALTER ATTRIBUTE a TYPE bigint;  ALTER TYPE comp RENAME ATTRIBUTE a TO b;
ALTER DOMAIN d SET DEFAULT x;   ALTER DOMAIN d DROP DEFAULT;   ALTER DOMAIN d SET|DROP NOT NULL;
ALTER DOMAIN d ADD CONSTRAINT c CHECK (...);    ALTER DOMAIN d DROP CONSTRAINT c;
ALTER DOMAIN d VALIDATE CONSTRAINT c;           ALTER DOMAIN d RENAME TO d2;

ALTER FUNCTION f(int) RENAME TO g;              ALTER FUNCTION f(int) OWNER TO r;
ALTER FUNCTION f(int) SET SCHEMA s;             ALTER FUNCTION f(int) IMMUTABLE PARALLEL SAFE SECURITY DEFINER;
ALTER FUNCTION f(int) SET search_path = public; ALTER FUNCTION f(int) DEPENDS ON EXTENSION e;
ALTER PROCEDURE p(int) RENAME TO q;             ALTER ROUTINE r(int) OWNER TO u;
ALTER AGGREGATE a(type) RENAME TO b;            ALTER OPERATOR +(int,int) OWNER TO r;
ALTER OPERATOR CLASS oc USING btree RENAME TO oc2;
ALTER OPERATOR FAMILY of USING btree ADD OPERATOR 1 < (int,int);
ALTER COLLATION c REFRESH VERSION;              ALTER COLLATION c RENAME TO c2;
ALTER CONVERSION c RENAME TO c2;
ALTER LANGUAGE l RENAME TO l2;                  ALTER LARGE OBJECT 123 OWNER TO r;
ALTER STATISTICS st SET STATISTICS 500;         ALTER STATISTICS st RENAME TO st2;
ALTER TRIGGER trg ON t RENAME TO trg2;          ALTER TRIGGER trg ON t DEPENDS ON EXTENSION e;
ALTER RULE r ON t RENAME TO r2;
ALTER EVENT TRIGGER et ENABLE|DISABLE|ENABLE REPLICA|ENABLE ALWAYS;
ALTER TEXT SEARCH CONFIGURATION cfg ALTER MAPPING FOR asciiword WITH english_stem;
ALTER TEXT SEARCH DICTIONARY d (StopWords = 'x');
ALTER TEXT SEARCH PARSER p RENAME TO p2;        ALTER TEXT SEARCH TEMPLATE t RENAME TO t2;
```

## Extension / FDW / replication / system
```sql
ALTER EXTENSION e UPDATE [TO 'v'];              ALTER EXTENSION e SET SCHEMA s;
ALTER EXTENSION e ADD|DROP TABLE t;
ALTER FOREIGN DATA WRAPPER w OPTIONS (ADD a 'b');   ALTER FOREIGN DATA WRAPPER w OWNER TO r;
ALTER SERVER srv OPTIONS (SET host 'h');        ALTER SERVER srv VERSION '2';
ALTER USER MAPPING FOR u SERVER srv OPTIONS (SET password 'x');
ALTER FOREIGN TABLE ft ADD COLUMN c int;        ALTER FOREIGN TABLE ft OPTIONS (SET table_name 'x');
ALTER PUBLICATION pub ADD TABLE t | DROP TABLE t | SET TABLE t1, t2 | SET (publish='insert');
ALTER PUBLICATION pub RENAME TO p2;             ALTER PUBLICATION pub OWNER TO r;
ALTER SUBSCRIPTION sub ENABLE|DISABLE;          ALTER SUBSCRIPTION sub REFRESH PUBLICATION;
ALTER SUBSCRIPTION sub CONNECTION '...';        ALTER SUBSCRIPTION sub SET PUBLICATION p [WITH (refresh=false)];
ALTER SUBSCRIPTION sub ADD|DROP PUBLICATION p;  ALTER SUBSCRIPTION sub SKIP (lsn = '0/1234');
ALTER SUBSCRIPTION sub SET (slot_name = 'x');   ALTER SUBSCRIPTION sub RENAME TO s2;

ALTER SYSTEM SET param = value;                 -- writes postgresql.auto.conf
ALTER SYSTEM RESET param;    ALTER SYSTEM RESET ALL;
SELECT pg_reload_conf();
```
