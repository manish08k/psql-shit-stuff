# 02 — CREATE (all objects)

## Database / schema / tablespace
```sql
CREATE DATABASE db [WITH] [OWNER r] [TEMPLATE t] [ENCODING 'UTF8'] [LOCALE_PROVIDER libc|icu|builtin]
  [LC_COLLATE 'x'] [LC_CTYPE 'x'] [LOCALE 'x'] [TABLESPACE ts] [ALLOW_CONNECTIONS true] [CONNECTION LIMIT n] [IS_TEMPLATE false]
  [STRATEGY wal_log|file_copy];
CREATE SCHEMA s [AUTHORIZATION role];
CREATE SCHEMA IF NOT EXISTS s;
CREATE TABLESPACE ts [OWNER r] LOCATION '/path';
```

## Table
```sql
CREATE [TEMP|UNLOGGED] TABLE [IF NOT EXISTS] t (
  id        bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  id2       int GENERATED BY DEFAULT AS IDENTITY (START 1 INCREMENT 1),
  name      text NOT NULL DEFAULT 'x',
  email     text UNIQUE,
  age       int CHECK (age >= 0),
  fk_id     int REFERENCES other(id) ON DELETE CASCADE ON UPDATE NO ACTION,
  total     numeric GENERATED ALWAYS AS (a + b) STORED,
  created   timestamptz DEFAULT now(),
  CONSTRAINT uq UNIQUE (a, b),
  CONSTRAINT ck CHECK (a < b),
  CONSTRAINT fk FOREIGN KEY (x) REFERENCES o(y) DEFERRABLE INITIALLY DEFERRED,
  EXCLUDE USING gist (room WITH =, during WITH &&)
);
CREATE TABLE t (LIKE src INCLUDING ALL);
CREATE TABLE child () INHERITS (parent);
CREATE TABLE tt OF composite_type;
CREATE TABLE t (...) WITH (fillfactor=70) TABLESPACE ts;
CREATE TABLE t (...) ON COMMIT DROP;                 -- TEMP tables: PRESERVE ROWS | DELETE ROWS | DROP
```
ON DELETE/UPDATE actions: `NO ACTION | RESTRICT | CASCADE | SET NULL | SET DEFAULT`

### Partitioning
```sql
CREATE TABLE m (id int, d date) PARTITION BY RANGE (d);
CREATE TABLE m_2026 PARTITION OF m FOR VALUES FROM ('2026-01-01') TO ('2027-01-01');
CREATE TABLE l (...) PARTITION BY LIST (region);
CREATE TABLE l_eu PARTITION OF l FOR VALUES IN ('eu','uk');
CREATE TABLE h (...) PARTITION BY HASH (id);
CREATE TABLE h0 PARTITION OF h FOR VALUES WITH (MODULUS 4, REMAINDER 0);
CREATE TABLE def PARTITION OF m DEFAULT;
```

### CREATE TABLE AS / SELECT INTO
```sql
CREATE TABLE t2 AS SELECT * FROM t WHERE x > 1;
CREATE TABLE t2 AS TABLE t WITH NO DATA;
CREATE TEMP TABLE t2 AS SELECT ...;
SELECT * INTO t2 FROM t;
```

## Index
```sql
CREATE [UNIQUE] INDEX [CONCURRENTLY] [IF NOT EXISTS] idx ON t (col);
CREATE INDEX idx ON t USING btree|hash|gist|spgist|gin|brin (col [ASC|DESC] [NULLS FIRST|LAST]);
CREATE INDEX idx ON t (a, b) INCLUDE (c);                      -- covering
CREATE INDEX idx ON t (lower(email));                          -- expression
CREATE INDEX idx ON t (col) WHERE active;                      -- partial
CREATE INDEX idx ON t (col text_pattern_ops);
CREATE INDEX idx ON t USING gin (jsonb_col jsonb_path_ops);
CREATE INDEX idx ON t (col) WITH (fillfactor=80) TABLESPACE ts;
CREATE UNIQUE INDEX idx ON t (a) NULLS NOT DISTINCT;
```

```sql
  [WITH CHECK OPTION | WITH LOCAL CHECK OPTION];
```

## Types / domains
```sql
CREATE TYPE mood AS ENUM ('sad','ok','happy');
CREATE TYPE pt AS (x int, y int);                              -- composite
CREATE TYPE r AS RANGE (SUBTYPE = int4);
CREATE TYPE t (INPUT = f_in, OUTPUT = f_out, ...);             -- base type
CREATE DOMAIN d AS text CHECK (VALUE ~ '^[a-z]+$') [NOT NULL] [DEFAULT 'x'];
CREATE CAST (src AS dst) WITH FUNCTION f(src) [AS ASSIGNMENT|AS IMPLICIT];
CREATE COLLATION c (provider = icu, locale = 'en-US');
CREATE CONVERSION c FOR 'LATIN1' TO 'UTF8' FROM fn;
CREATE TRANSFORM FOR type LANGUAGE plpython3u (FROM SQL WITH FUNCTION f1, TO SQL WITH FUNCTION f2);
```

## Functions / procedures / aggregates / operators
```sql
CREATE [OR REPLACE] FUNCTION f(a int, b int DEFAULT 0) RETURNS int
  LANGUAGE sql|plpgsql|... [IMMUTABLE|STABLE|VOLATILE] [STRICT] [PARALLEL SAFE]
  [SECURITY DEFINER] [COST n] [SET search_path = public]
  AS $$ SELECT a + b $$;
CREATE FUNCTION f() RETURNS TABLE (id int, name text) AS $$ ... $$ LANGUAGE sql;
CREATE FUNCTION f() RETURNS trigger AS $$ BEGIN ... RETURN NEW; END $$ LANGUAGE plpgsql;
CREATE FUNCTION f(...) RETURNS SETOF t ...;
CREATE [OR REPLACE] PROCEDURE p(a int) LANGUAGE plpgsql AS $$ BEGIN ... END $$;
CREATE AGGREGATE agg (type) (SFUNC = f, STYPE = type, [FINALFUNC = ff], [INITCOND = '0']);
CREATE OPERATOR === (LEFTARG = t, RIGHTARG = t, FUNCTION = f, [COMMUTATOR = ===]);
CREATE OPERATOR CLASS oc [DEFAULT] FOR TYPE t USING btree AS OPERATOR 1 <, ..., FUNCTION 1 cmp(t,t);
CREATE OPERATOR FAMILY of USING btree;
CREATE LANGUAGE lang [HANDLER h];                              -- (use CREATE EXTENSION normally)
```

## Triggers / rules / event triggers
```sql
CREATE [OR REPLACE] [CONSTRAINT] TRIGGER trg
  { BEFORE | AFTER | INSTEAD OF } { INSERT | UPDATE [OF col] | DELETE | TRUNCATE } [OR ...]
  ON t [REFERENCING OLD TABLE AS o NEW TABLE AS n]
  FOR EACH ROW|STATEMENT [WHEN (cond)]
  EXECUTE FUNCTION fn(args);
CREATE RULE r AS ON INSERT TO t DO INSTEAD NOTHING;
CREATE EVENT TRIGGER et ON ddl_command_start|ddl_command_end|table_rewrite|sql_drop
  [WHEN TAG IN ('CREATE TABLE')] EXECUTE FUNCTION fn();
```

## Security
```sql
CREATE ROLE r [WITH] LOGIN|NOLOGIN SUPERUSER|NOSUPERUSER CREATEDB CREATEROLE REPLICATION BYPASSRLS
  INHERIT CONNECTION LIMIT n PASSWORD 'pw' VALID UNTIL 'ts' IN ROLE r2 ROLE r3 ADMIN r4;
CREATE USER u PASSWORD 'pw';                                   -- = CREATE ROLE ... LOGIN
CREATE GROUP g;                                                -- = CREATE ROLE
CREATE POLICY p ON t [AS PERMISSIVE|RESTRICTIVE] [FOR ALL|SELECT|INSERT|UPDATE|DELETE]
  [TO role] USING (expr) [WITH CHECK (expr)];
```

## Extensions / FDW / replication / misc
```sql

CREATE USER MAPPING FOR u SERVER srv OPTIONS (user 'x', password 'y');
CREATE FOREIGN TABLE ft (id int, name text) SERVER srv OPTIONS (schema_name 'public', table_name 't');
IMPORT FOREIGN SCHEMA remote_schema [LIMIT TO (t1) | EXCEPT (t2)] FROM SERVER srv INTO local_schema;
CREATE PUBLICATION pub FOR TABLE t1, t2 | FOR ALL TABLES | FOR TABLES IN SCHEMA s [WITH (publish='insert,update')];
CREATE SUBSCRIPTION sub CONNECTION 'host=... dbname=... user=...' PUBLICATION pub [WITH (copy_data=true, slot_name='x')];
CREATE STATISTICS st (dependencies, ndistinct, mcv) ON a, b FROM t;
CREATE TEXT SEARCH CONFIGURATION cfg (COPY = english);
CREATE TEXT SEARCH DICTIONARY d (TEMPLATE = snowball, language = 'english');
CREATE TEXT SEARCH PARSER p (START=..., GETTOKEN=..., END=..., LEXTYPES=...);
CREATE TEXT SEARCH TEMPLATE tpl (INIT=..., LEXIZE=...);
```
