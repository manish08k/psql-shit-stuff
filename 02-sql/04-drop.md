# 04 — DROP (all objects)

Common modifiers: `IF EXISTS`, `CASCADE` (drop dependents), `RESTRICT` (default, refuse if dependents).

```sql
DROP DATABASE [IF EXISTS] db [WITH (FORCE)];     -- FORCE terminates connections
DROP SCHEMA [IF EXISTS] s [CASCADE];
DROP TABLE [IF EXISTS] t1, t2 [CASCADE];
DROP INDEX [CONCURRENTLY] [IF EXISTS] idx;
DROP VIEW [IF EXISTS] v [CASCADE];
DROP MATERIALIZED VIEW [IF EXISTS] mv [CASCADE];
DROP SEQUENCE [IF EXISTS] seq;
DROP TABLESPACE [IF EXISTS] ts;

DROP TYPE [IF EXISTS] t [CASCADE];
DROP DOMAIN [IF EXISTS] d [CASCADE];
DROP CAST [IF EXISTS] (src AS dst);
DROP COLLATION [IF EXISTS] c;
DROP CONVERSION [IF EXISTS] c;
DROP TRANSFORM [IF EXISTS] FOR type LANGUAGE lang;

DROP FUNCTION [IF EXISTS] f(int, text) [CASCADE];
DROP PROCEDURE [IF EXISTS] p(int);
DROP ROUTINE [IF EXISTS] r(int);
DROP AGGREGATE [IF EXISTS] a(type);
DROP OPERATOR [IF EXISTS] +(int, int);
DROP OPERATOR CLASS [IF EXISTS] oc USING btree;
DROP OPERATOR FAMILY [IF EXISTS] of USING btree;
DROP LANGUAGE [IF EXISTS] l;

DROP TRIGGER [IF EXISTS] trg ON t [CASCADE];
DROP RULE [IF EXISTS] r ON t;
DROP EVENT TRIGGER [IF EXISTS] et;
DROP POLICY [IF EXISTS] p ON t;

DROP ROLE [IF EXISTS] r;
DROP USER [IF EXISTS] u;
DROP GROUP [IF EXISTS] g;
DROP OWNED BY role [CASCADE];                    -- drop everything owned + revoke privileges
REASSIGN OWNED BY old_role TO new_role;          -- run before DROP ROLE

DROP EXTENSION [IF EXISTS] e [CASCADE];
DROP ACCESS METHOD [IF EXISTS] am;
DROP FOREIGN DATA WRAPPER [IF EXISTS] w [CASCADE];
DROP SERVER [IF EXISTS] srv [CASCADE];
DROP USER MAPPING [IF EXISTS] FOR u SERVER srv;
DROP FOREIGN TABLE [IF EXISTS] ft;
DROP PUBLICATION [IF EXISTS] pub;
DROP SUBSCRIPTION [IF EXISTS] sub;               -- ALTER SUBSCRIPTION sub DISABLE; SET (slot_name=NONE) first if remote unreachable
DROP STATISTICS [IF EXISTS] st;

DROP TEXT SEARCH CONFIGURATION [IF EXISTS] cfg;
DROP TEXT SEARCH DICTIONARY [IF EXISTS] d;
DROP TEXT SEARCH PARSER [IF EXISTS] p;
DROP TEXT SEARCH TEMPLATE [IF EXISTS] t;
```

## Safe role-removal recipe
```sql
REASSIGN OWNED BY old_role TO new_role;
DROP OWNED BY old_role;
DROP ROLE old_role;
```
